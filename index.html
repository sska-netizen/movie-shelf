export default { async fetch(request, env) { const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, OPTIONS", "Access-Control-Allow-Headers": "Content-Type" }; if (request.method === "OPTIONS") { return new Response(null, { headers: corsHeaders }); } const url = new URL(request.url); if (url.pathname !== "/search") { return new Response( JSON.stringify({ message: "Movie Shelf API is working!" }), { headers: { "Content-Type": "application/json", ...corsHeaders } } ); } const query = url.searchParams.get("q"); const page = Math.max( 1, parseInt(url.searchParams.get("page") || "1", 10) ); if (!query) { return new Response( JSON.stringify({ error: "検索ワードがありません" }), { status: 400, headers: { "Content-Type": "application/json", ...corsHeaders } } ); } /* 検索文字の揺れを軽減 トイ・ストーリー トイ ストーリー トイ　ストーリー ↓ トイストーリー */ const normalizedQuery = query .normalize("NFKC") .replace(/[\s・･·•]/g, "") .trim(); /* TMDBから検索する関数 */ async function searchTMDB(searchQuery, searchPage) { const tmdbUrl = "https://api.themoviedb.org/3/search/movie" + "?api_key=" + encodeURIComponent(env.TMDB_API_KEY) + "&query=" + encodeURIComponent(searchQuery) + "&language=ja-JP" + "®ion=JP" + "&include_adult=false" + "&page=" + searchPage; const response = await fetch(tmdbUrl); if (!response.ok) { throw new Error( "TMDB error: " + response.status ); } return await response.json(); } try { /* 通常検索 */ const data = await searchTMDB(query, page); /* 検索文字を正規化した検索も実行 */ let normalizedData = { results: [], total_pages: 0, total_results: 0 }; if ( normalizedQuery && normalizedQuery !== query ) { normalizedData = await searchTMDB( normalizedQuery, page ); } /* 2つの検索結果を統合 同じTMDB IDは1つにする */ const combined = [ ...(data.results || []), ...(normalizedData.results || []) ]; const uniqueMovies = []; const seen = new Set(); for (const movie of combined) { if (seen.has(movie.id)) { continue; } seen.add(movie.id); uniqueMovies.push({ id: movie.id, title: movie.title || "", originalTitle: movie.original_title || "", releaseDate: movie.release_date || "", overview: movie.overview || "", posterPath: movie.poster_path ? "https://image.tmdb.org/t/p/w500" + movie.poster_path : null }); } /* 検索結果 20件まで返す */ const movies = uniqueMovies.slice(0, 20); /* 次ページがあるか */ const hasNextPage = page < Math.max( data.total_pages || 0, normalizedData.total_pages || 0 ); return new Response( JSON.stringify({ results: movies, page: page, hasNextPage: hasNextPage, totalResults: Math.max( data.total_results || 0, normalizedData.total_results || 0 ) }), { headers: { "Content-Type": "application/json; charset=UTF-8", ...corsHeaders } } ); } catch (error) { return new Response( JSON.stringify({ error: "TMDB検索に失敗しました", detail: error.message }), { status: 500, headers: { "Content-Type": "application/json", ...corsHeaders } } ); } } };
